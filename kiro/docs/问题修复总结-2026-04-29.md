### 1. 数据绑定

```vue
<!-- 正确 -->
<text>{{ baziStore.currentBaziData.year_pillar.shishen || '-' }}</text>

<!-- 错误 -->
<text>{{ baziStore.currentBaziData.month_pillar.shishen }}</text>  <!-- 绑定了错误的柱子 -->
```

### 2. 字段兜底

```vue
<!-- 单个字段 -->
<text>{{ pillar.shishen || '-' }}</text>

<!-- 数组字段 -->
<text v-for="(ss, index) in (pillar.canggan_shishen || [])" :key="index">
  {{ ss || '-' }}
</text>
<text v-if="!pillar.canggan_shishen || pillar.canggan_shishen.length === 0">-</text>
```
